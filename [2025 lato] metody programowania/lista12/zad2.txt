
Pokaż, że dla dowolnej listy xs zachodzi rev (rev xs) = xs.

let rec rev xs = 
    match xs with 
    | [] -> []
    | x :: xs -> (rev xs) @ [x]

let rec @ l1 l2 = 
    match l1 with
    | [] -> l2
    | x :: xs -> x :: (x2 @ l2)
